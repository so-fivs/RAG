# Comandos para ollama si no les abrió instalado o subido (copy y paste) 
# Instalar y levantar Docker
sudo yum install docker -y
sudo systemctl start docker
sudo usermod -aG docker ec2-user
newgrp docker

# Levantar contenedor Ollama con volumen persistente
docker run -d \
  --name ollama \
  -p 11434:11434 \
  -v /home/ec2-user/SageMaker/ollama_data:/root/.ollama \
  --restart unless-stopped \
  ollama/ollama

# Pull modelos necesarios
docker exec ollama ollama pull nomic-embed-text
docker exec ollama ollama pull qwen2.5:1.5b

# Alias útiles
echo "alias ollama-logs='docker logs -f ollama'" >> ~/.bashrc
echo "alias ollama-restart='docker restart ollama'" >> ~/.bashrc
echo "alias ollama-models='docker exec ollama ollama list'" >> ~/.bashrc
source ~/.bashrc
